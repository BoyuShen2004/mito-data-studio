# Core + the AI-assist stack, CPU-only builds.
#
# Enables EfficientSAM Point/Box Mask and SAM2 tracking without a GPU. Both run
# noticeably slower than the CUDA path — this profile exists so a machine with
# no NVIDIA driver can still exercise the full feature set, not as the
# recommended way to annotate at volume.
#
# Same package versions as requirements-ai-gpu.txt; only the wheel index and
# the ONNX Runtime package differ.

-r requirements-core.txt

# CPU wheels. Without this index pip resolves the default (CUDA) build of
# torch from PyPI, which pulls ~5 GB of nvidia-* wheels the profile exists to
# avoid.
--extra-index-url https://download.pytorch.org/whl/cpu

torch==2.5.1
torchvision==0.20.1

# CPU ONNX Runtime. Never install this alongside onnxruntime-gpu — the two
# packages provide the same module and the last one installed silently wins.
onnxruntime==1.26.0

# SAM2 helpers.
hydra-core==1.3.4
iopath==0.1.10
tqdm==4.69.1
